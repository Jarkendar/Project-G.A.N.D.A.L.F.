# Pinned for reproducibility; the same pins as imladris-rag/pyproject.toml.
# The embedding models themselves are pinned separately, to fixed Hugging Face
# Hub revisions in imladris-rag/imladris/models.py (MODEL_REGISTRY;
# BILBO_EMBED_REVISION overrides) — independent of these
# library versions, so a future `pip install` here can never silently change
# which model weights get downloaded.
#
# torch is deliberately left unpinned: it's a transitive dependency of
# sentence-transformers, and hard-pinning it here risks an unresolvable
# version conflict with sentence-transformers' own internal bounds. Let pip
# resolve whichever torch version sentence-transformers==5.6.0 requires.
sentence-transformers==5.6.0
numpy==2.5.0
